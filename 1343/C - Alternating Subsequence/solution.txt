def main():
    t = int(input())
    for _ in range(t):
        n = int(input())
        arr = list(map(int,input().split()))
        max_holder = arr[0]
        res = 0
        for i in range(1,len(arr)):
            if (arr[i-1] > 0 and arr[i] < 0) or (arr[i-1]  < 0 and arr[i] > 0):
                res += max_holder
                max_holder = arr[i]
            elif (arr[i-1] > 0 and arr[i] > 0) or (arr[i-1]  < 0 and arr[i] < 0):
                max_holder = max(max_holder,arr[i])
        res += max_holder      
        print(res)
    
main()
