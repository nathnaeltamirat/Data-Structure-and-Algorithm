t = int(input())for _ in range(t):    n = int(input())    nums = list(map(int,input().split()))    high_pos = float('-inf')    high_neg = float('-inf')    l = 0    r = 0    res = 0    while r < n:        if nums[r]  > 0 and nums[l] < 0:            res += high_neg            high_pos = nums[r]            high_neg = float('-inf')            l = r        elif nums[r]  < 0 and nums[l] > 0:            res += high_pos            high_pos = float('-inf')            high_neg = nums[r]            l = r        elif nums[l] > 0 and nums[l] > 0:            high_pos = max(high_pos,nums[r])        elif nums[l] < 0 and nums[r] < 0:            high_neg = max(high_neg,nums[r])        r += 1    if high_neg != float('-inf'):        res += high_neg    if high_pos != float('-inf'):        res += high_pos    print(res)
