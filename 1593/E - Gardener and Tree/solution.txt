from collections import defaultdict,deque t = int(input())for _ in range(t):    input()    graph = defaultdict(list)    indegree = defaultdict(int)    q = deque()    n,k = map(int,input().split())     for _ in range(n-1):        a, b = map(int,input().split())        graph[a].append(b)        graph[b].append(a)        indegree[a] += 1        indegree[b] += 1    if n <= 2 and k >= 1:        print(0)        continue    for i in range(1,n+1):        if indegree[i] == 1:            q.append(i)    size = n    op = 0    while q:        length = len(q)        if op == k:            break        for _ in range(length):            node = q.popleft()            for neigh in graph[node]:                indegree[neigh] -= 1                if indegree[neigh] == 1:                    q.append(neigh)        op += 1        size-=length    print(size)
